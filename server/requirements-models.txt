# ===========================================================================
# Dub server - the GPU side of the main venv.
#
# Install after server/requirements.txt:
#
#   pip install -r server/requirements.txt
#   pip install -r server/requirements-models.txt --extra-index-url \
#       https://download.pytorch.org/whl/cu121
#
# numpy is pinned to 1.26.4 because LatentSync needs it. That pin is the
# whole reason the subtitle remover runs in its own venv: paddle over there
# wants numpy 2.2, and the two ABIs do not mix.
#
# Model files are not downloaded by pip:
#   LatentSync/checkpoints/latentsync_unet.pt      (~1.3 GB)
#   LatentSync/checkpoints/whisper/tiny.pt
# Run ./download_models.sh once, before the first job.
# ===========================================================================

-r ../LatentSync/requirements.txt

# Pinned, and pinned here rather than in VoxCPM/requirements.txt, which
# only says "voxcpm". Unpinned, pip is free to backtrack onto the 1.x line,
# and 1.x still loads the VoxCPM2 weights but has no reference_wav_path on
# generate(): the voice cloning call raises TypeError, and anything that
# works around the TypeError is no longer cloning. That is what made the
# first Colab run sound wrong. 2.x also restores the device argument on
# from_pretrained().
voxcpm==2.0.3

torch==2.5.1
torchvision==0.20.1
# demucs pulls torchaudio in on its own and picks a build that does not
# match the torch above. Pinned here so it is installed from the cu121
# index in the same call.
torchaudio==2.5.1
numpy==1.26.4

# Reads the speech. The desktop tool ran this in a child process because
# ctranslate2 next to Qt crashed the app; there is no Qt here.
faster-whisper==1.1.1

# Splits voice from music.
demucs==4.1.0

# Writes the wav files VoxCPM returns as arrays.
soundfile

# The Hugging Face fast transfer backend. Without it the VoxCPM2 download
# is slow enough to look like a hang.
hf_xet
